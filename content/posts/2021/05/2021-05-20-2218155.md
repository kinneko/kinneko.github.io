　M5StickCに歩行用のサンプルスケッチを書き込んでしまうと、PuppyCに充電する場合には、動き続けることになってしまう。M5StickCの左のボタンを長押しにして電源を切ってから、PuppyCの電源を入れて充電できる。

　動かしているとM5StickCがズレて、コネクタの差し込みが甘くなっていくので、ゴムをかける方向を変えた。LCDも見やすくなった。しかし、ポリウレタンの弱点で、切れやすいので翌日には切れてしまったよ...

![](/2021-05-20/2218155-01.png)

　分割のテストする。

```cpp
int split(String data, char delimiter, String *dst){
  int index = 0;
  int arraySize = (sizeof(data))/sizeof((data[0]));
  int datalength = data.length();
  for(int i = 0; i < datalength; i++){
    char tmp = data.charAt(i);
    if( tmp == delimiter ){
      index++;
      if( index > (arraySize - 1)) return -1;
    }
    else dst[index] += tmp;
  }
  return (index + 1);
}

void setup() {
  Serial.begin(115200);
}

void loop() {
  if (Serial.available() > 0) {
    String serStrings = Serial.readString();
    serStrings.trim();
    Serial.println(serStrings);
    String serStringsList[4];
    split(serStrings, ',', serStringsList);
    Serial.println(serStringsList[0]);
    Serial.println(serStringsList[1]);
    Serial.println(serStringsList[2]);
    Serial.println(serStringsList[3]);
    Serial.println("END");
  }
}
```

　あとは、角度がそれぞれのサーボの絶対値になっているので、直感的にわかりやすいように相対値にしておきたい。

```cpp
#include <M5StickC.h>
#include "puppy.h"

void puppy_start() {
  angle_all_set(90,90,90,90); delay(100);
}

int split(String data, char delimiter, String *dst){
  int index = 0;
  int arraySize = (sizeof(data))/sizeof((data[0]));
  int datalength = data.length();
  for(int i = 0; i < datalength; i++){
    char tmp = data.charAt(i);
    if( tmp == delimiter ){
      index++;
      if( index > (arraySize - 1)) return -1;
    }
    else dst[index] += tmp;
  }
  return (index + 1);
}

void setup() {
  Serial.begin(115200);
  M5.begin();
  M5.Axp.ScreenBreath(10); //7-12 backlight
  M5.Lcd.setRotation(0); //0,1,2,3
  M5.Lcd.fillScreen(BLACK);
  M5.Lcd.fillCircle(40, 80, 30, RED);
  M5.Lcd.setTextColor(WHITE); M5.Lcd.setTextSize(2);
  M5.Lcd.setCursor(5, 60); M5.Lcd.printf("PuppyC");
  M5.Lcd.setCursor(5, 80); M5.Lcd.printf("Serial");
  IIC_Init(); //sda 0 scl 26
  angle_set(1,90); angle_set(2,90); angle_set(3,90); angle_set(4,90);
  delay(1000);
}

void loop() {
  if (Serial.available() > 0) {
    String serStrings = Serial.readString();
    serStrings.trim();
    Serial.println(serStrings);
    String serStringsList[4];
    split(serStrings, ',', serStringsList);

    int leg1 = serStringsList[0].toInt();
    int leg2 = 180 - serStringsList[1].toInt();
    int leg3 = serStringsList[2].toInt();
    int leg4 = 180 - serStringsList[3].toInt();
    Serial.println(serStringsList[0]+":"+leg1);
    Serial.println(serStringsList[1]+":"+leg2);
    Serial.println(serStringsList[2]+":"+leg3);
    Serial.println(serStringsList[3]+":"+leg4);
    angle_all_set(leg1,leg2,leg3,leg4);
  }
}
```
