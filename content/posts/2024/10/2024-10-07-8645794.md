　というわけで、こちらもAPIから情報を取れるようにする。デバイスのリストと、それぞれのデバイスの持っているファンクションのリストが取れればいいかな。JSONとかはライブラリが重くなりがちなので入れないでおく。

　とりあえず、ルートパスにアクセスがあったときにはデバイスリストを返し、デバイス名だけのときにはファンクションのリストを返すようにする。

　サーバーのルートハンドラを追加する。

```text
void setupServer() {
  server.on("/", handleRootRequest); // ルートパスのハンドラを登録
  server.onNotFound(handleDeviceRequest);
  server.begin();
  Serial.println("HTTP server started");
}
```

　handleRootRequest()関数を追加する。

```text
void handleRootRequest() {
  // デバイスのリストを文字列として作成
  String deviceList = "Devices: ";
  for (int i = 0; i < sizeof(devices)/sizeof(devices[0]); i++) {
    deviceList += devices[i].name;
    if (i < sizeof(devices)/sizeof(devices[0]) - 1) {
      deviceList += ", ";
    }
  }
  // レスポンスを送信
  server.send(200, "text/plain", deviceList);
}
```

　handleDeviceRequest()関数を修正する。デバイス名のみのときの処理を追加して、デバイス名が指定されていない場合の処理を削除する。

```text
void handleDeviceRequest() {
  String path = server.uri(); // リクエストされたPATHを取得
  Serial.println("Requested path: " + path);

  // パスをスラッシュで分割してデバイス名とアクション名を取得
  int firstSlash = path.indexOf('/', 1);
  if (firstSlash == -1) {
    server.send(400, "text/plain", "Invalid URL format. Use /device or /device/action");
    return;
  }

  String deviceName = path.substring(1, firstSlash);
  String actionName = path.substring(firstSlash + 1);

  // デバイスを検索
  bool deviceFound = false;
  for (int i = 0; i < sizeof(devices)/sizeof(devices[0]); i++) {
    if (deviceName.equalsIgnoreCase(devices[i].name)) {
      deviceFound = true;
      bool actionFound = false;

      if (actionName.length() == 0) {
        // アクション名が指定されていない場合、デバイスの機能リストを返す
        String functionsList = "Functions: ";
        for (int j = 0; j < devices[i].functionCount; j++) {
          functionsList += devices[i].functions[j].name;
          if (j < devices[i].functionCount - 1) {
            functionsList += ", ";
          }
        }
        server.send(200, "text/plain", functionsList);
        return;
      } else {
        // 機能を検索
        bool actionFound = false;
        for (int j = 0; j < devices[i].functionCount; j++) {
          if (actionName.equalsIgnoreCase(devices[i].functions[j].name)) {
            actionFound = true;
            // IRコードを送信
            uint64_t code = devices[i].functions[j].irCode.code;
            uint16_t bits = devices[i].functions[j].irCode.bits;
            sendIR(NEC, code, bits, 1, 1);
            server.send(200, "text/plain", "Signal '" + actionName + "' sent to " + deviceName);
            Serial.printf("送信したNECコード: %#llx, ビット数: %d\n", code, bits);
            break;
          }
        }

        if (!actionFound) {
          server.send(400, "text/plain", "Action '" + actionName + "' not found for device '" + deviceName + "'.");
        }
        return;
      }
    }
  }
  if (!deviceFound) {
    server.send(404, "text/plain", "Device '" + deviceName + "' not found.");
  }
}
```
