　簡単に受信してシリアルに出すスケッチ。

```
#include "M5StickCPlus2.h"
#include <IRrecv.h>
#include <IRutils.h>

// 赤外線受信ピン
const uint16_t RECV_PIN = 33;

// IR受信オブジェクトを作成
IRrecv irrecv(RECV_PIN);
decode_results results;

void setup() {
  M5.begin();
  Serial.begin(115200);
  irrecv.enableIRIn();  // IR受信を開始
}

void loop() {
  // 赤外線信号を受信した場合
  if (irrecv.decode(&results)) {
    // 受信内容をシリアルモニタに表示
    Serial.println(resultToHumanReadableBasic(&results));
    // 受信した信号を再度準備
    irrecv.resume();
  }
  delay(100);
}
```

#include "M5StickCPlus2.h"

// set the input pin.  设置引脚
int ir_recv_pin = 33;
int ir_send_pin = 32;

int last_recv_value = 0;
int cur_recv_value  = 0;

void setup() {
    auto cfg = M5.config();
    StickCP2.begin(cfg);
    StickCP2.Display.setRotation(1);
    pinMode(ir_recv_pin, INPUT);
    pinMode(ir_send_pin, OUTPUT);
    digitalWrite(ir_send_pin, 1);
    StickCP2.Display.setTextColor(GREEN);
    StickCP2.Display.setTextSize(2);
    StickCP2.Display.setCursor(0, 0);
    StickCP2.Display.print("Test for IR receiver: ");
}

void loop() {
    cur_recv_value = digitalRead(ir_recv_pin);
    if (last_recv_value != cur_recv_value) {
        StickCP2.Display.setCursor(0, 25);
        StickCP2.Display.fillRect(0, 25, 150, 25, BLACK);
        if (cur_recv_value ==
            0) {  // 0: detected 1: not detected,  0检测到,1没有检测到
            StickCP2.Display.print("detected!");
        }
        last_recv_value = cur_recv_value;
    }
    Serial.println(cur_recv_value);
    delay(1000);
}

　改良する。

```
#include "M5StickCPlus2.h"

// set the input pin.
int ir_recv_pin = 33;
int ir_send_pin = 32;

unsigned long last_recv_value = 0;
unsigned long cur_recv_value = 0;
bool ir_received = false;

void IRAM_ATTR handleIR() {
    static unsigned long ir_value = 0;
    static unsigned long lastTime = 0;
    unsigned long now = micros();
    unsigned long duration = now - lastTime;
    lastTime = now;

    if (duration > 10000) { // Assuming 10ms gap indicates a new signal
        if (ir_value != 0) {
            cur_recv_value = ir_value;
            ir_received = true;
        }
        ir_value = 0;
    } else {
        ir_value = (ir_value << 1) | (duration > 500); // Assuming 500us threshold for 1 or 0
    }
}

void setup() {
    auto cfg = M5.config();
    StickCP2.begin(cfg);
    StickCP2.Display.setRotation(1);
    pinMode(ir_recv_pin, INPUT);
    pinMode(ir_send_pin, OUTPUT);
    digitalWrite(ir_send_pin, 1);
    StickCP2.Display.setTextColor(GREEN);
    StickCP2.Display.setTextSize(2);
    StickCP2.Display.setCursor(0, 0);
    StickCP2.Display.print("Test for IR receiver: ");

    attachInterrupt(ir_recv_pin, handleIR, CHANGE);
}

void loop() {
    if (ir_received) {
        ir_received = false;
        StickCP2.Display.setCursor(0, 25);
        StickCP2.Display.fillRect(0, 25, 150, 25, BLACK);
        StickCP2.Display.print("IR signal: ");
        StickCP2.Display.print(cur_recv_value, HEX);
        Serial.print("Received IR signal: ");
        Serial.println(cur_recv_value, HEX);
    }
    delay(1000);
}
```
