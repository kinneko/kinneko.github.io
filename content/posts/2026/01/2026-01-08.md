　メインのコードを書く。

% vi main/mdns_check.c

#include <string.h>

#include "freertos/FreeRTOS.h"

#include "freertos/task.h"

#include "esp_log.h"

#include "esp_event.h"

#include "nvs_flash.h"

#include "esp_netif.h"

#include "esp_wifi.h"

#include "mdns.h"

　static const char *TAG = "mdns_check";

　static void start_mdns(void)

　{

　ESP_ERROR_CHECK(mdns_init());

// hostname.local になる

　ESP_ERROR_CHECK(mdns_hostname_set(CONFIG_MDNS_HOSTNAME));

　ESP_ERROR_CHECK(mdns_instance_name_set("ESP-IDF mDNS Check"));

// 動作確認用：HTTPサービスを広告（実サーバは立てなくてもOK）

　ESP_ERROR_CHECK(mdns_service_add(NULL, "_http", "_tcp", 80, NULL, 0));

// TXTレコード（任意）

　mdns_txt_item_t txt[] = {

　{"board", "esp32"},

　{"app", "mdns_check"},

　};

　ESP_ERROR_CHECK(mdns_service_txt_set("_http", "_tcp", txt, 2));

　ESP_LOGI(TAG, "mDNS started: %s.local", CONFIG_MDNS_HOSTNAME);

　ESP_LOGI(TAG, "Service: _http._tcp on port 80");

　}

　static void wifi_event_handler(void *arg, esp_event_base_t event_base,

　int32_t event_id, void *event_data)

　{

　(void)arg;

　(void)event_data;

　if (event_base == WIFI_EVENT && event_id == WIFI_EVENT_STA_START) {

　ESP_LOGI(TAG, "Wi-Fi start -> connect");

　esp_wifi_connect();

　} else if (event_base == WIFI_EVENT && event_id == WIFI_EVENT_STA_DISCONNECTED) {

　ESP_LOGW(TAG, "Wi-Fi disconnected -> reconnect");

　esp_wifi_connect();

　} else if (event_base == IP_EVENT && event_id == IP_EVENT_STA_GOT_IP) {

　ESP_LOGI(TAG, "Got IP -> start mDNS");

　start_mdns();

　}

　}

　static void wifi_init_sta(void)

　{

　esp_netif_create_default_wifi_sta();
