　ESP-IDFを v6.1.0にしたので続き。

　5GHzのみ、ch36でAPを起動する。

```text
% find ./
./
.//CMakeLists.txt
.//sdkconfig.defaults
.//main
.//main/CMakeLists.txt
.//main/idf_component.yml
.//main/main.c

% cat CMakeLists.txt
cmake_minimum_required(VERSION 3.16)
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(stampc5_5g_portal)

% cat sdkconfig.defaults
CONFIG_IDF_TARGET="esp32c5"
CONFIG_ESPTOOLPY_FLASHSIZE_4MB=y
CONFIG_ESP_WIFI_SOFTAP_SUPPORT=y
CONFIG_ESP_CONSOLE_USB_SERIAL_JTAG=y
CONFIG_ESP_CONSOLE_SECONDARY_NONE=y

% cat main/CMakeLists.txt
idf_component_register(SRCS "main.c" INCLUDE_DIRS "." REQUIRES esp_wifi esp_netif esp_event esp_http_server nvs_flash lwip mdns)

% cat main/idf_component.yml
dependencies:
espressif/mdns: "^1.14.0"

% cat main/main.c
```
