　Daydream Controller は BLE の FE55 サービスにある notify characteristic「00000001-1000-1000-8000-00805f9b34fb」を購読すれば、20バイトの生データを取得できるそうな。まず、ペアリングして情報が取れるか確認する。

```python
% vi daydream.py
#!/usr/bin/env python3

import asyncio
from bleak import BleakScanner, BleakClient

DEVICE_NAME = "Daydream controller"
CHAR_UUID = "00000001-1000-1000-8000-00805f9b34fb"


def notification_handler(sender, data: bytearray):
    print(data.hex(" "))


async def main():
    print("Scanning for Daydream controller...")

    device = await BleakScanner.find_device_by_name(
        DEVICE_NAME,
        timeout=20.0,
    )

    if device is None:
        print("Daydream controller not found")
        return

    print(f"Found: {device.name} ({device.address})")
    print("Connecting...")

    async with BleakClient(device) as client:
        print("Connected")

        await client.start_notify(
            CHAR_UUID,
            notification_handler,
        )

        print("Receiving notifications. Ctrl-C to stop.")

        while True:
            await asyncio.sleep(1)


try:
    asyncio.run(main())
except KeyboardInterrupt:
    print("\nStopped")
```

　コントローラのHomeボタンを長押ししてLEDを点滅状態にしてから、以下を実行。

　20バイトのNotificationをデコードしてリアルタイム表示するように変更する。

```python
#!/usr/bin/env python3

import asyncio
from dataclasses import dataclass

from bleak import BleakClient, BleakScanner


DEVICE_NAME = "Daydream controller"
CHAR_UUID = "00000001-1000-1000-8000-00805f9b34fb"


@dataclass
class DaydreamPacket:
    timestamp: int
    seq: int

    orientation_x: int
    orientation_y: int
    orientation_z: int

    accel_x: int
    accel_y: int
    accel_z: int

    gyro_x: int
    gyro_y: int
    gyro_z: int

    touch_x: int
    touch_y: int

    volume_up: bool
    volume_down: bool
    app: bool
    home: bool
    trackpad_click: bool

    mystery: int


class BitReader:
    def __init__(self, data: bytes):
        self.value = int.from_bytes(data, "big")
        self.total_bits = len(data) * 8
        self.pos = 0

    def get(self, width: int) -> int:
        shift = self.total_bits - self.pos - width
        value = (self.value >> shift) & ((1 << width) - 1)
        self.pos += width
        return value

    def get_signed(self, width: int) -> int:
        value = self.get(width)
        sign = 1 << (width - 1)
        if value & sign:
            value -= 1 << width
        return value


def decode_packet(data: bytes) -> DaydreamPacket:
    if len(data) != 20:
        raise ValueError(f"unexpected packet size: {len(data)}")

    r = BitReader(data)

    timestamp = r.get(9)
    seq = r.get(5)

    # Wire order is X, Z, Y.
    orientation_x = r.get_signed(13)
    orientation_z = r.get_signed(13)
    orientation_y = r.get_signed(13)

    accel_x = r.get_signed(13)
    accel_z = r.get_signed(13)
    accel_y = r.get_signed(13)

    gyro_x = r.get_signed(13)
    gyro_z = r.get_signed(13)
    gyro_y = r.get_signed(13)

    touch_x = r.get(8)
    touch_y = r.get(8)

    volume_up = bool(r.get(1))
    volume_down = bool(r.get(1))
    app = bool(r.get(1))
    home = bool(r.get(1))
    trackpad_click = bool(r.get(1))

    mystery = r.get(8)

    return DaydreamPacket(
        timestamp=timestamp,
        seq=seq,

        orientation_x=orientation_x,
        orientation_y=orientation_y,
        orientation_z=orientation_z,

        accel_x=accel_x,
        accel_y=accel_y,
        accel_z=accel_z,

        gyro_x=gyro_x,
        gyro_y=gyro_y,
        gyro_z=gyro_z,

        touch_x=touch_x,
        touch_y=touch_y,

        volume_up=volume_up,
        volume_down=volume_down,
        app=app,
        home=home,
        trackpad_click=trackpad_click,

        mystery=mystery,
    )


def buttons_to_string(p: DaydreamPacket) -> str:
    buttons = []

    if p.trackpad_click:
        buttons.append("TOUCH_CLICK")
    if p.home:
        buttons.append("HOME")
    if p.app:
        buttons.append("APP")
    if p.volume_down:
        buttons.append("VOL-")
    if p.volume_up:
        buttons.append("VOL+")

    return ",".join(buttons) if buttons else "-"


def notification_handler(sender, data: bytearray):
    try:
        p = decode_packet(bytes(data))
    except ValueError as e:
        print(f"RX error: {e}: {bytes(data).hex(' ')}")
        return

    touching = p.touch_x != 0 or p.touch_y != 0

    print(
        f"t={p.timestamp:3d} "
        f"seq={p.seq:2d}  "
        f"ori=({p.orientation_x:5d},{p.orientation_y:5d},{p.orientation_z:5d})  "
        f"acc=({p.accel_x:5d},{p.accel_y:5d},{p.accel_z:5d})  "
        f"gyro=({p.gyro_x:5d},{p.gyro_y:5d},{p.gyro_z:5d})  "
        f"touch=({p.touch_x:3d},{p.touch_y:3d}) "
        f"{'TOUCH' if touching else '     '}  "
        f"buttons={buttons_to_string(p):<12} "
        f"unk=0x{p.mystery:02x}"
    )


async def main():
    print("Scanning for Daydream controller...")
    print("Hold the controller HOME button until the LED pulses.")

    device = await BleakScanner.find_device_by_name(
        DEVICE_NAME,
        timeout=30.0,
    )

    if device is None:
        print("Daydream controller not found.")
        return

    print(f"Found: {device.name}")
    print(f"ID:    {device.address}")
    print("Connecting...")

    async with BleakClient(device, timeout=30.0) as client:
        print("Connected.")
        print(f"Subscribing: {CHAR_UUID}")

        await client.start_notify(
            CHAR_UUID,
            notification_handler,
        )

        print("Receiving data. Press Ctrl-C to stop.")
        print()

        try:
            await asyncio.Event().wait()
        finally:
            try:
                await client.stop_notify(CHAR_UUID)
            except Exception:
                pass


if __name__ == "__main__":
    try:
        asyncio.run(main())
    except KeyboardInterrupt:
        print("\nStopped.")
```
