　Lチカしかしてないので、ちょっとC5っぽいことをしてみる。

　ピンが生えてないので、いろいろめんどくさいけど、とりあえず、5GHz WiFiで接続できるかどうか試す。残念なのは、2.4/5GHz対応は二択で両立できないこと。

　試すだけなら、とりあえず、5GHzでAP接続してキャプティブポータルを起動できるようにして、接続先を選んでパスワードを入れたらSTA接続して... ん？ とれたアドレスはどうやって知るかな。mDNSでもしとく？

　んで、HTTPサーバーを起動して、アクセスしたらなんか画面に出すくらいでいいかな。接続状態でも出しとくか。

　前回懲りたので、ESP-IDF 6.0.2を使う。

　AIさんに書いてもらう。

static bool mdns_started = false;

static void start_mdns(void) {
if (mdns_started) return;
ESP_ERROR_CHECK(mdns_init());
ESP_ERROR_CHECK(mdns_hostname_set("c5"));
ESP_ERROR_CHECK(mdns_instance_name_set("Stamp C5 5GHz Wi-Fi Test"));
ESP_ERROR_CHECK(mdns_service_add(NULL, "_http", "_tcp", 80, NULL, 0));
mdns_started = true;
ESP_LOGI(TAG, "mDNS: http://c5.local/ (HTTP service _http._tcp)");
}


static bool load_credentials(void) {
nvs_handle_t h;
if (nvs_open("wifi_test", NVS_READONLY, &h) != ESP_OK) return false;
size_t sl = sizeof(saved_ssid), pl = sizeof(saved_pass);
esp_err_t a = nvs_get_str(h, "ssid", saved_ssid, &sl);
esp_err_t b = nvs_get_str(h, "pass", saved_pass, &pl);
nvs_close(h);
return a == ESP_OK && b == ESP_OK && saved_ssid[0];
}
static esp_err_t save_credentials(const char *ssid, const char *pass) {
nvs_handle_t h;
esp_err_t e = nvs_open("wifi_test", NVS_READWRITE, &h);
if (e != ESP_OK) return e;
if ((e = nvs_set_str(h, "ssid", ssid)) == ESP_OK)
e = nvs_set_str(h, "pass", pass);
if (e == ESP_OK) e = nvs_commit(h);
nvs_close(h);
return e;
}
static void reboot_task(void *arg) {
vTaskDelay(pdMS_TO_TICKS(1600));
esp_restart();
}
static esp_err_t reply_html(httpd_req_t *req, const char *html) {
httpd_resp_set_type(req, "text/html; charset=utf-8");
httpd_resp_set_hdr(req, "Cache-Control", "no-store");
return httpd_resp_sendstr(req, html);
}
static void url_decode(char *s) {
char *dst = s;
for (char *src = s; *src; src++) {
if (*src == '+') *dst++ = ' ';
else if (*src == '%' && isxdigit((unsigned char)src[1]) && isxdigit((unsigned char)src[2])) {
char hex[3] = {src[1], src[2], 0};
*dst++ = (char)strtol(hex, NULL, 16); src += 2;
} else *dst++ = *src;
}
*dst = 0;
}
static bool form_field(const char *body, const char *name, char *out, size_t cap) {
size_t nl = strlen(name);
const char *p = body;
while (*p) {
if (!strncmp(p, name, nl) && p[nl] == '=') {
p += nl + 1;
const char *end = strchr(p, '&');
size_t n = end ? (size_t)(end - p) : strlen(p);
if (n >= cap) return false;
memcpy(out, p, n); out[n] = 0; url_decode(out); return true;
}
p = strchr(p, '&'); if (!p) break; p++;
}
return false;
}
static esp_err_t page_get(httpd_req_t *req) {
if (provisioning) {
return reply_html(req,
"<!doctype html><html><head><meta name='viewport' content='width=device-width,initial-scale=1'>"
"<meta charset='utf-8'><title>StampC5 Wi-Fi setup</title></head>"

"<p>SSID<br><input name='ssid' maxlength='32' required style='width:100%'></p>"
"<p>Password<br><input name='password' type='password' maxlength='64' style='width:100%'></p>"
"<button type='submit'>保存して再起動</button></form>"
"<p>設定AP: " SETUP_SSID " / <a href='http://192.168.4.1/'>192.168.4.1</a></p>"
"</body></html>");
}
char ip[16] = "not assigned", gw[16] = "-";
esp_netif_ip_info_t info;
if (sta_has_ip && esp_netif_get_ip_info(sta_netif, &info) == ESP_OK) {
snprintf(ip, sizeof(ip), IPSTR, IP2STR(&info.ip));
snprintf(gw, sizeof(gw), IPSTR, IP2STR(&info.gw));
}
wifi_ap_record_t ap = {0};
bool linked = esp_wifi_sta_get_ap_info(&ap) == ESP_OK;
char html[1400];
snprintf(html, sizeof(html),
"<!doctype html><html><head><meta name='viewport' content='width=device-width,initial-scale=1'>"
"<meta charset='utf-8'><meta http-equiv='refresh' content='5'></head>"
"<body style='font:16px system-ui;max-width:36em;margin:2em auto;padding:1em'>"
"<h2>StampC5 5GHz connection</h2>"
"<p>Status: <b>%s</b></p><p>SSID: %s</p><p>IP: %s</p><p>Gateway: %s</p>"
"<p>Channel: %u (%s)</p><p>RSSI: %d dBm</p>"
"<p><a href='/status'>JSON status</a></p>"
"<form action='/reset' method='post'><button>Wi-Fi設定を消去</button></form></body></html>",
sta_has_ip ? "GOT IP" : "connecting", saved_ssid, ip, gw,
linked ? ap.primary : 0, linked && ap.primary > 14 ? "5GHz" : "not 5GHz", linked ? ap.rssi : 0);
return reply_html(req, html);
}
static esp_err_t status_get(httpd_req_t *req) {
esp_netif_ip_info_t info;
char ip[16] = "";
if (sta_has_ip && esp_netif_get_ip_info(sta_netif, &info) == ESP_OK)
snprintf(ip, sizeof(ip), IPSTR, IP2STR(&info.ip));
wifi_ap_record_t ap = {0};
bool linked = esp_wifi_sta_get_ap_info(&ap) == ESP_OK;
char json[384];
snprintf(json, sizeof(json),
"{\"connected\":%s,\"ssid\":\"%s\",\"ip\":\"%s\",\"channel\":%u,\"band\":\"%s\",\"rssi\":%d}\n",
sta_has_ip ? "true" : "false", saved_ssid, ip,
linked ? ap.primary : 0, linked && ap.primary > 14 ? "5GHz" : "unknown", linked ? ap.rssi : 0);
httpd_resp_set_type(req, "application/json");
return httpd_resp_sendstr(req, json);
}
static esp_err_t save_post(httpd_req_t *req) {
if (!provisioning) return httpd_resp_send_err(req, HTTPD_403_FORBIDDEN, "not in setup mode");
if (req->content_len == 0 || req->content_len > 512)
return httpd_resp_send_err(req, HTTPD_400_BAD_REQUEST, "invalid input length");
char body[513], ssid[97], password[193];
int got = 0;
while (got < req->content_len) {
int n = httpd_req_recv(req, body + got, req->content_len - got);
if (n <= 0) return ESP_FAIL;
got += n;
}
body[got] = 0;
if (!form_field(body, "ssid", ssid, sizeof(ssid)) ||
!form_field(body, "password", password, sizeof(password)))
return httpd_resp_send_err(req, HTTPD_400_BAD_REQUEST, "missing fields");
size_t sl = strlen(ssid), pl = strlen(password);
if (sl < 1 || sl > 32 || pl > 64 || (pl > 0 && pl < 8))
return httpd_resp_send_err(req, HTTPD_400_BAD_REQUEST, "SSID/password length invalid");
esp_err_t e = save_credentials(ssid, password);
if (e != ESP_OK) return httpd_resp_send_err(req, HTTPD_500_INTERNAL_SERVER_ERROR, "NVS failure");
reply_html(req, "<html><meta charset='utf-8'><body><h2>保存しました。再起動します。</h2><p>シリアルログで5GHz接続結果を確認してください。</p></body></html>");
xTaskCreate(reboot_task, "reboot", 2048, NULL, 5, NULL);
return ESP_OK;
}
static esp_err_t reset_post(httpd_req_t *req) {
nvs_handle_t h;
if (nvs_open("wifi_test", NVS_READWRITE, &h) != ESP_OK)
return httpd_resp_send_err(req, HTTPD_500_INTERNAL_SERVER_ERROR, "NVS failure");
esp_err_t e = nvs_erase_all(h);
if (e == ESP_OK) e = nvs_commit(h);
nvs_close(h);
if (e != ESP_OK) return httpd_resp_send_err(req, HTTPD_500_INTERNAL_SERVER_ERROR, "NVS failure");
reply_html(req, "<html><body>Settings cleared. Rebooting...</body></html>");
xTaskCreate(reboot_task, "reboot", 2048, NULL, 5, NULL);
return ESP_OK;
}
static esp_err_t not_found(httpd_req_t *req, httpd_err_code_t err) {
(void)err;
if (!provisioning) return httpd_resp_send_err(req, HTTPD_404_NOT_FOUND, "Not found");
httpd_resp_set_status(req, "302 Found");
httpd_resp_set_hdr(req, "Location", "http://192.168.4.1/");
return httpd_resp_sendstr(req, "Open http://192.168.4.1/");
}
static void start_http(void) {
httpd_config_t config = HTTPD_DEFAULT_CONFIG();
config.lru_purge_enable = true;
ESP_ERROR_CHECK(httpd_start(&server, &config));
const httpd_uri_t root = {.uri="/", .method=HTTP_GET, .handler=page_get};
const httpd_uri_t save = {.uri="/save", .method=HTTP_POST, .handler=save_post};
const httpd_uri_t reset = {.uri="/reset", .method=HTTP_POST, .handler=reset_post};
const httpd_uri_t status = {.uri="/status", .method=HTTP_GET, .handler=status_get};
ESP_ERROR_CHECK(httpd_register_uri_handler(server, &root));
ESP_ERROR_CHECK(httpd_register_uri_handler(server, &save));
ESP_ERROR_CHECK(httpd_register_uri_handler(server, &reset));
ESP_ERROR_CHECK(httpd_register_uri_handler(server, &status));
ESP_ERROR_CHECK(httpd_register_err_handler(server, HTTPD_404_NOT_FOUND, not_found));
}
/* Catch-all DNS A responder. This is intentionally only active in setup/AP mode. */
static void dns_task(void *arg) {
(void)arg;
int fd = socket(AF_INET, SOCK_DGRAM, IPPROTO_IP);
if (fd < 0) { ESP_LOGE(TAG, "DNS socket failed"); vTaskDelete(NULL); return; }
struct sockaddr_in addr = {.sin_family=AF_INET, .sin_port=htons(DNS_PORT), .sin_addr.s_addr=htonl(INADDR_ANY)};
if (bind(fd, (struct sockaddr *)&addr, sizeof(addr)) < 0) {
ESP_LOGE(TAG, "DNS bind failed"); close(fd); vTaskDelete(NULL); return;
}
uint8_t buf[512];
while (true) {
struct sockaddr_storage peer;
socklen_t peerlen = sizeof(peer);
int n = recvfrom(fd, buf, sizeof(buf), 0, (struct sockaddr *)&peer, &peerlen);
if (n < 17) continue;
/* Parse DNS QNAME safely, accepting only one uncompressed question. */
int p = 12;
while (p < n && buf[p] && (buf[p] & 0xc0) == 0 && p + 1 + buf[p] < n) p += buf[p] + 1;
if (p >= n || buf[p] != 0 || p + 5 > n) continue;
int qend = p + 5;
uint16_t qtype = ((uint16_t)buf[p+1] << 8) | buf[p+2];
uint16_t qclass = ((uint16_t)buf[p+3] << 8) | buf[p+4];
if (buf[2] & 0x80 || buf[4] != 0 || buf[5] != 1 || qclass != 1) continue;
/* Answer A requests; for other types return NOERROR with zero answers. */
buf[2] = 0x81; buf[3] = 0x80;
buf[6] = 0; buf[7] = qtype == 1 ? 1 : 0;
memset(buf+8, 0, 4);
int total = qend;
if (qtype == 1 && total + 16 <= (int)sizeof(buf)) {
uint8_t answer[] = {0xc0,0x0c,0x00,0x01,0x00,0x01,0,0,0,30,0,4,192,168,4,1};
memcpy(buf+total, answer, sizeof(answer)); total += sizeof(answer);
}
sendto(fd, buf, total, 0, (struct sockaddr *)&peer, peerlen);
}
}
static void wifi_events(void *arg, esp_event_base_t base, int32_t id, void *data) {
(void)arg;
if (base == WIFI_EVENT && id == WIFI_EVENT_STA_DISCONNECTED) {
wifi_event_sta_disconnected_t *e = data;
sta_has_ip = false;
ESP_LOGW(TAG, "Disconnected reason=%u; reconnecting", e->reason);
esp_wifi_connect();
} else if (base == WIFI_EVENT && id == WIFI_EVENT_STA_CONNECTED) {
wifi_event_sta_connected_t *e = data;
ESP_LOGI(TAG, "Associated: channel=%u (%s)", e->channel, e->channel > 14 ? "5GHz" : "2.4GHz");
} else if (base == IP_EVENT && id == IP_EVENT_STA_GOT_IP) {
ip_event_got_ip_t *e = data;
sta_has_ip = true;
start_mdns();
ESP_LOGI(TAG, "5GHz STA IP: " IPSTR " -> http://" IPSTR "/", IP2STR(&e->ip_info.ip), IP2STR(&e->ip_info.ip));
}
}
void app_main(void) {
esp_err_t e = nvs_flash_init();
if (e == ESP_ERR_NVS_NO_FREE_PAGES || e == ESP_ERR_NVS_NEW_VERSION_FOUND) {
ESP_ERROR_CHECK(nvs_flash_erase());
e = nvs_flash_init();
}
ESP_ERROR_CHECK(e);
ESP_ERROR_CHECK(esp_netif_init());
ESP_ERROR_CHECK(esp_event_loop_create_default());
provisioning = !load_credentials();
if (provisioning) esp_netif_create_default_wifi_ap();
else sta_netif = esp_netif_create_default_wifi_sta();
wifi_init_config_t cfg = WIFI_INIT_CONFIG_DEFAULT();
ESP_ERROR_CHECK(esp_wifi_init(&cfg));
ESP_ERROR_CHECK(esp_wifi_set_storage(WIFI_STORAGE_RAM));
if (provisioning) {
ESP_ERROR_CHECK(esp_wifi_set_mode(WIFI_MODE_AP));
wifi_config_t ap = {0};
strcpy((char *)ap.ap.ssid, SETUP_SSID);
strcpy((char *)ap.ap.password, SETUP_PASS);
ap.ap.ssid_len = strlen(SETUP_SSID);
ap.ap.channel = 1;
ap.ap.max_connection = 4;
ap.ap.authmode = WIFI_AUTH_WPA2_PSK;
ESP_ERROR_CHECK(esp_wifi_set_config(WIFI_IF_AP, &ap));
ESP_ERROR_CHECK(esp_wifi_start());
ESP_LOGI(TAG, "Setup AP: %s / password: %s", SETUP_SSID, SETUP_PASS);
ESP_LOGI(TAG, "Open http://192.168.4.1/");
start_http();
xTaskCreate(dns_task, "dns", 4096, NULL, 4, NULL);
} else {
ESP_ERROR_CHECK(esp_event_handler_register(WIFI_EVENT, ESP_EVENT_ANY_ID, wifi_events, NULL));
ESP_ERROR_CHECK(esp_event_handler_register(IP_EVENT, IP_EVENT_STA_GOT_IP, wifi_events, NULL));
ESP_ERROR_CHECK(esp_wifi_set_mode(WIFI_MODE_STA));
wifi_config_t sta = {0};
memcpy(sta.sta.ssid, saved_ssid, strlen(saved_ssid));
memcpy(sta.sta.password, saved_pass, strlen(saved_pass));
ESP_ERROR_CHECK(esp_wifi_set_config(WIFI_IF_STA, &sta));
ESP_ERROR_CHECK(esp_wifi_start());
ESP_ERROR_CHECK(esp_wifi_set_band_mode(WIFI_BAND_MODE_5G_ONLY));
ESP_ERROR_CHECK(esp_wifi_connect());
ESP_LOGI(TAG, "5GHz-only mode, connecting to SSID: %s", saved_ssid);
start_http();
}
}
