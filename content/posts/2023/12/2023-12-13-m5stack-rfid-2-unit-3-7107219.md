　モジュールのどちらの面からでも、カードやタグは問題なく認識している。

　スケッチでは、setupの中でserial.begin()を実行していないけど、これはM5.begin()の中でやっているのかな。シリアルの速度設定がわからないけど、115kでいいようだ。

　シリアルポートのほうにも結果が表示されている。

![](/2023-12-13/7107219-01.png)

　現状では、loop()のたびにIDを表示している。毎秒２、３回表示される。最初のカードを認識してから１分経つまで同じカードは表示しないようにしたい。また、画面をスクロールするか、今のUIDのみを大きく表示するようにしたい。

　そんな方針で、サンプルスケッチを書き換える。今回は、少し賢くなったChatGPT4に指示だけ出してスケッチを生成させてみる。それを微調整した。

　データブロックのほうをシリアルにダンプするようにしてみる。

　スケッチはこんな感じ。

~~~cpp
#include <M5StickCPlus.h>
#include <Wire.h>
#include "MFRC522_I2C.h"

MFRC522 mfrc522(0x28);  // Create MFRC522 instance.
unsigned long lastCardReadTime = 0;  // Last time a card was read
byte lastCardUID[10];  // Array to store the last card's UID
byte lastCardUIDSize = 0;  // Size of the last card's UID

void setup() {
    M5.begin();
    M5.Lcd.setRotation(3);
    M5.Lcd.fillScreen(BLACK);
    M5.Lcd.setTextSize(3);  // Increase text size for better visibility
    M5.Lcd.setCursor(0, 2);
    M5.Lcd.println("RFID UNIT 2");
    Wire.begin();         // Initialize I2C
    mfrc522.PCD_Init();   // Init MFRC522
    ShowReaderDetails();  // Show details of PCD - MFRC522 Card Reader details
    Serial.println(F("Scan PICC to see UID, type, and data blocks..."));
    M5.Lcd.println("Scan PICC...");
}

void loop() {
    // Look for new cards
    if (!mfrc522.PICC_IsNewCardPresent() || !mfrc522.PICC_ReadCardSerial()) {
        delay(50);
        return;
    }

    // Check if the card UID is the same as the last one and if 1 minute has not passed
    if (isSameCardUID(mfrc522.uid.uidByte, mfrc522.uid.size) && (millis() - lastCardReadTime < 60000)) {
        return;  // If it's the same card and less than 1 minute has passed, do nothing
    }

    // Save the new card UID and the read time
    memcpy(lastCardUID, mfrc522.uid.uidByte, mfrc522.uid.size);
    lastCardUIDSize = mfrc522.uid.size;
    lastCardReadTime = millis();

    // Display the new card UID
    M5.Lcd.fillScreen(BLACK);  // Clear the screen before displaying new UID
    M5.Lcd.setCursor(10, 2);   // Reset cursor position
    Serial.print(F("Card UID:"));
    for (byte i = 0; i < mfrc522.uid.size; i++) {
        Serial.print(mfrc522.uid.uidByte[i] < 0x10 ? " 0" : " ");
        Serial.print(mfrc522.uid.uidByte[i], HEX);
        M5.Lcd.print(mfrc522.uid.uidByte[i] < 0x10 ? " 0" : " ");
        M5.Lcd.print(mfrc522.uid.uidByte[i], HEX);
    }
    Serial.println();
    M5.Lcd.println();  // Move to the next line on the LCD
}

bool isSameCardUID(byte *uid, byte size) {
    if (size != lastCardUIDSize) return false;
    for (byte i = 0; i < size; i++) {
        if (uid[i] != lastCardUID[i]) return false;
    }
    return true;
}

void ShowReaderDetails() {
    // Get the MFRC522 software version
    byte v = mfrc522.PCD_ReadRegister(mfrc522.VersionReg);
    Serial.print(F("MFRC522 Software Version: 0x"));
    Serial.print(v, HEX);
    if (v == 0x91)
        Serial.print(F(" = v1.0"));
    else if (v == 0x92)
        Serial.print(F(" = v2.0"));
    else
        Serial.print(F(" (unknown)"));
    Serial.println("");
    // When 0x00 or 0xFF is returned, communication probably failed
    if ((v == 0x00) || (v == 0xFF)) {
        Serial.println(
            F("WARNING: Communication failure, is the MFRC522 properly "
              "connected?"));
    }
}
~~~
